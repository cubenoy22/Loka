cmake_minimum_required(VERSION 3.18)
project(LokaQuickJSProbe LANGUAGES C CXX)

# Standalone probe: intentionally outside the application/Scene build graph.
include(${CMAKE_CURRENT_LIST_DIR}/../../../example/SmirkyCard/cmake/QuickJS.cmake)
if(CMAKE_SYSTEM_NAME STREQUAL "Retro68")
  find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
  set(probe_source "${CMAKE_CURRENT_BINARY_DIR}/classic-source")
  execute_process(
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/prepare_source.py"
      "${smirkycard_quickjs_SOURCE_DIR}" "${probe_source}"
    RESULT_VARIABLE patch_result)
  if(NOT patch_result EQUAL 0)
    message(FATAL_ERROR "Classic probe source preparation failed")
  endif()
  get_target_property(probe_engine_sources smirkycard_quickjs SOURCES)
  set(probe_patched_sources)
  foreach(engine_source IN LISTS probe_engine_sources)
    get_filename_component(engine_filename "${engine_source}" NAME)
    list(APPEND probe_patched_sources "${probe_source}/${engine_filename}")
  endforeach()
  set_property(TARGET smirkycard_quickjs PROPERTY SOURCES ${probe_patched_sources})
  set_property(TARGET smirkycard_quickjs PROPERTY INCLUDE_DIRECTORIES "${probe_source}")
  set_property(TARGET smirkycard_quickjs PROPERTY INTERFACE_INCLUDE_DIRECTORIES "${probe_source}")
  set_property(TARGET smirkycard_quickjs PROPERTY INTERFACE_SYSTEM_INCLUDE_DIRECTORIES "${probe_source}")
endif()
target_compile_options(smirkycard_quickjs PRIVATE -Os -ffunction-sections -fdata-sections)
if(CMAKE_SYSTEM_NAME STREQUAL "Retro68")
  add_application(LokaQuickJSProbe main.c CONSOLE)
  target_link_options(LokaQuickJSProbe PRIVATE -Wl,--mac-strip-macsbug)
else()
  add_executable(LokaQuickJSProbe main.c)
endif()
set_target_properties(LokaQuickJSProbe PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
target_compile_options(LokaQuickJSProbe PRIVATE -Wall -Wextra -Werror -Os -ffunction-sections -fdata-sections)
target_link_options(LokaQuickJSProbe PRIVATE -Wl,--gc-sections)
target_link_libraries(LokaQuickJSProbe smirkycard_quickjs)
if(CMAKE_SYSTEM_NAME STREQUAL "Retro68")
  target_compile_definitions(smirkycard_quickjs PRIVATE LOKA_QUICKJS_68K_PROBE=1)
  target_compile_definitions(LokaQuickJSProbe PRIVATE LOKA_QUICKJS_68K_PROBE=1)
  target_sources(LokaQuickJSProbe PRIVATE clock.c memory.c Size.r)
endif()
